/* ───────── 初期セットアップ（1回だけ手動実行） ───────── */

function setup() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  Object.keys(SHEET).filter(k => HEADERS[k]).forEach(k => ensureSheet_(ss, SHEET[k], HEADERS[k]));

  const master = ss.getSheetByName(SHEET.MASTER);
  if (master.getLastRow() < 2) {
    const rows = [];
    SEED_TREE.forEach(c => {
      const p = /^[QSCA]/.test(c.cat) ? c.cat[0] : 'K'; let n = 0;
      Object.keys(c.subs).forEach(sub => {
        const items = c.subs[sub];
        if (!items.length) rows.push([p + '000-' + sub, c.cat, c.color, sub, '', 1, '', false, rows.length + 1]);
        items.forEach(t => { n++; rows.push([p + ('00' + n).slice(-3), c.cat, c.color, sub, t, 1, '', true, rows.length + 1]); });
      });
    });
    master.getRange(2, 1, rows.length, HEADERS.MASTER.length).setValues(rows);
  }
  master.getRange('H2:H').insertCheckboxes();

  const stores = ss.getSheetByName(SHEET.STORES);
  if (stores.getLastRow() < 2) {
    const rows = [];
    Object.keys(SEED_STORE_TREE).forEach(g => SEED_STORE_TREE[g].forEach(x => rows.push([x[0], g, x[1], true])));
    stores.getRange(2, 1, rows.length, 4).setValues(rows);
  }
  stores.getRange('D2:D').insertCheckboxes();

  const props = PropertiesService.getScriptProperties();
  if (!props.getProperty('PHOTO_FOLDER_ID')) {
    const folder = DriveApp.createFolder('QSCチェック写真');
    props.setProperty('PHOTO_FOLDER_ID', folder.getId());
  }
  if (!props.getProperty('EDIT_PIN')) props.setProperty('EDIT_PIN', '0000');

  formatSheets_();
  buildDashboard_();
  SpreadsheetApp.getUi && Logger.log('セットアップ完了。編集用PINは スクリプトプロパティ EDIT_PIN（初期値 0000）で変更してください。');
}

function ensureSheet_(ss, name, headers) {
  let sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  if (sh.getLastRow() === 0) {
    sh.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold').setBackground('#e8eeec');
    sh.setFrozenRows(1);
  }
  return sh;
}

/* ───────── アプリから呼ばれる関数 ───────── */

function getConfig() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const master = readRows_(ss.getSheetByName(SHEET.MASTER)).map(r => ({
    id: String(r['ID']),
    cat: String(r['カテゴリ']),
    color: String(r['カテゴリ色'] || '#555555'),
    sub: String(r['サブカテゴリ']),
    text: String(r['項目'] || ''),
    weight: Number(r['配点']) || 1,
    brand: String(r['対象ブランド'] || ''),
    active: r['有効'] === true || String(r['有効']).toUpperCase() === 'TRUE',
    order: Number(r['並び順']) || 0,
  })).filter(r => r.id && r.cat);
  master.sort((a, b) => a.order - b.order);

  const stores = readRows_(ss.getSheetByName(SHEET.STORES))
    .filter(r => r['店舗名'] && (r['有効'] === true || String(r['有効']).toUpperCase() === 'TRUE'))
    .map(r => ({ name: String(r['店舗名']), genre: String(r['業態'] || 'その他'), brand: String(r['ブランド'] || '') }));

  return { master, stores, demo: false };
}

/** 項目マスタを丸ごと書き換える（削除は「有効=FALSE」で残すので過去データは壊れない） */
function saveMaster(pin, items) {
  checkPin_(pin);
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET.MASTER);
    const rows = items.map((r, i) => [r.id, r.cat, r.color, r.sub, r.text, Number(r.weight) || 1, r.brand || '', !!r.active, i + 1]);
    if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, HEADERS.MASTER.length).clearContent();
    if (rows.length) sh.getRange(2, 1, rows.length, HEADERS.MASTER.length).setValues(rows);
    sh.getRange('H2:H').insertCheckboxes();
    return { ok: true, count: rows.length };
  } finally {
    lock.releaseLock();
  }
}

/** 1項目だけ書き換える（文面・配点・有効）。他の人の編集を上書きしないよう該当行だけ更新 */
function updateItem(pin, id, patch) {
  checkPin_(pin);
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET.MASTER);
    const n = sh.getLastRow() - 1;
    const ids = n > 0 ? sh.getRange(2, 1, n, 1).getValues().flat().map(String) : [];
    const k = ids.indexOf(String(id));
    if (k < 0) throw new Error('この項目がスプシに見つかりません。「項目を読み込み直す」をしてください');
    const row = k + 2;
    if ('text' in patch) sh.getRange(row, 5).setValue(String(patch.text));
    if ('weight' in patch) sh.getRange(row, 6).setValue(Number(patch.weight) || 1);
    if ('active' in patch) sh.getRange(row, 8).setValue(!!patch.active);
    return { ok: true };
  } finally {
    lock.releaseLock();
  }
}

function checkPin(pin) {
  checkPin_(pin);
  return true;
}

function checkPin_(pin) {
  const want = PropertiesService.getScriptProperties().getProperty('EDIT_PIN') || '0000';
  if (String(pin) !== String(want)) throw new Error('PINが違います');
}

/** 写真を1枚ずつ受け取って Drive に保存 */
function uploadPhoto(recordId, folderName, fileName, dataUrl) {
  const folder = recordFolder_(recordId, folderName);
  const m = String(dataUrl).match(/^data:(.+?);base64,(.*)$/);
  if (!m) throw new Error('写真データの形式が不正です');
  const blob = Utilities.newBlob(Utilities.base64Decode(m[2]), m[1], fileName);
  const file = folder.createFile(blob);
  return { url: file.getUrl(), folderUrl: folder.getUrl() };
}

function recordFolder_(recordId, folderName) {
  const root = DriveApp.getFolderById(PropertiesService.getScriptProperties().getProperty('PHOTO_FOLDER_ID'));
  const name = folderName + '_' + recordId;
  const it = root.getFoldersByName(name);
  return it.hasNext() ? it.next() : root.createFolder(name);
}

    const ch = headersFor_(cats, HEADERS.CATS);
    const catRows = p.cats.map(c => { const o = { '記録ID': p.recordId, 'チェック日': m.date, '店舗': m.store, '業態': m.genre || '', 'ブランド': m.brand, 'カテゴリ': c.cat,
      '○': c.pass, '△': c.half || 0, '✕': c.fail, '対象外': c.na, '未チェック': c.open, '得点率': c.rate }; return ch.map(k => k in o ? o[k] : ''); });
    if (catRows.length) {
      const r0 = cats.getLastRow() + 1;
      cats.getRange(r0, 1, catRows.length, catRows[0].length).setValues(catRows);
      cats.getRange(r0, ch.indexOf('得点率') + 1, catRows.length, 1).setNumberFormat('0.0%');
    }

    const det = ss.getSheetByName(SHEET.DETAIL);
    const detRows = p.items.map(i => [p.recordId, m.date, m.store, m.brand, i.cat, i.sub, i.id, i.text, i.weight, i.result, i.comment, (i.photos || []).join('\n')]);
    if (detRows.length) det.getRange(det.getLastRow() + 1, 1, detRows.length, detRows[0].length).setValues(detRows);

    try { buildDashboard_(); } catch (e) { console.error('dashboard', e); }
    return { ok: true };
  } finally {
    lock.releaseLock();
  }
}

/** その店舗の直近の得点（前回比の表示用） */
function getHistory(store) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const recs = readRows_(ss.getSheetByName(SHEET.RECORDS)).filter(r => r['店舗'] === store);
  const last = recs.slice(-5);
  const ids = last.map(r => r['記録ID']);
  const cats = readRows_(ss.getSheetByName(SHEET.CATS)).filter(r => ids.indexOf(r['記録ID']) >= 0);
  return last.map(r => ({
    recordId: r['記録ID'],
    date: fmtDate_(r['チェック日']),
    checker: r['チェック者'],
    rate: Number(r['得点率']),
    cats: cats.filter(c => c['記録ID'] === r['記録ID']).map(c => ({ cat: c['カテゴリ'], rate: c['得点率'] === '' ? null : Number(c['得点率']) })),
  }));
}

// 判定の基準（平均得点率がこの値以上ならこの判定）。変えたい時はここを書き換える
const JUDGE = [[0.9, '◎ 優良'], [0.8, '○ 合格'], [0.7, '△ 要注意'], [0, '✕ 要改善']];

function onOpen() {
  SpreadsheetApp.getUi().createMenu('QSCチェック')
    .addItem('ダッシュボードを更新', 'refreshDashboard')
    .addItem('シートの見た目を整える', 'formatSheets')
    .addToUi();
}
function refreshDashboard() { buildDashboard_(); }
function formatSheets() { formatSheets_(); buildDashboard_(); }

function judge_(v) {
  if (v == null) return '';
  for (let i = 0; i < JUDGE.length; i++) if (v >= JUDGE[i][0]) return JUDGE[i][1];
  return '';
}
function num_(v) { return (v === '' || v == null || isNaN(Number(v))) ? null : Number(v); }
function avg_(a) { const b = a.filter(x => x != null); return b.length ? b.reduce((s, x) => s + x, 0) / b.length : null; }
function shortCat_(c) { const a = String(c).split(/[＝=]/)[0]; return a.length <= 3 ? a : String(c); }
function toDate_(d) { return d instanceof Date ? d : new Date(String(d).replace(/-/g, '/')); }
function widths_(sh, arr) { arr.forEach((w, i) => sh.setColumnWidth(i + 1, w)); }
function pctRule_(rg) {
  const T = SpreadsheetApp.InterpolationType.NUMBER;
  return SpreadsheetApp.newConditionalFormatRule()
    .setGradientMinpointWithValue('#f4c7c3', T, '0.6')
    .setGradientMidpointWithValue('#fce8b2', T, '0.8')
    .setGradientMaxpointWithValue('#b7e1cd', T, '1')
    .setRanges([rg]).build();
}
function textRule_(rg, text, bg, color) {
  return SpreadsheetApp.newConditionalFormatRule().whenTextContains(text).setBackground(bg).setFontColor(color).setRanges([rg]).build();
}

function buildDashboard_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(SHEET.DASH);
  if (!sh) sh = ss.insertSheet(SHEET.DASH, 0);
  const recs = readRows_(ss.getSheetByName(SHEET.RECORDS)).filter(r => r['記録ID']);
  const cats = readRows_(ss.getSheetByName(SHEET.CATS)).filter(r => r['記録ID']);
  const det = readRows_(ss.getSheetByName(SHEET.DETAIL)).filter(r => r['記録ID']);

  sh.clear();
  sh.clearConditionalFormatRules();
  sh.setHiddenGridlines(true);
  const rules = [];

  const catNames = [];
  cats.forEach(c => { if (catNames.indexOf(c['カテゴリ']) < 0) catNames.push(c['カテゴリ']); });
  const allMean = avg_(recs.map(r => num_(r['得点率'])));

  // 店舗別
  const byStore = {};
  recs.forEach(r => { (byStore[r['店舗']] = byStore[r['店舗']] || []).push(r); });
  const storeRows = Object.keys(byStore).map(st => {
    const rs = byStore[st].slice().sort((a, b) => (toDate_(a['チェック日']) - toDate_(b['チェック日'])) || (toDate_(a['送信日時']) - toDate_(b['送信日時'])));
    const ids = rs.map(r => r['記録ID']);
    const mean = avg_(rs.map(r => num_(r['得点率'])));
    const last = rs[rs.length - 1], prev = rs.length > 1 ? rs[rs.length - 2] : null;
    const catAvg = catNames.map(cn => {
      const v = avg_(cats.filter(c => c['カテゴリ'] === cn && ids.indexOf(c['記録ID']) >= 0).map(c => num_(c['得点率'])));
      return v == null ? '' : v;
    });
    const lr = num_(last['得点率']), pr = prev ? num_(prev['得点率']) : null;
    return [st, last['業態'] || '', rs.length, mean == null ? '' : mean, judge_(mean)].concat(catAvg, [fmtDate_(last['チェック日']), lr == null ? '' : lr, (lr != null && pr != null) ? lr - pr : '']);
  }).sort((a, b) => (b[3] === '' ? -1 : b[3]) - (a[3] === '' ? -1 : a[3]));

  // ✕が多い項目
  const it = {};
  det.forEach(d => {
    const res = d['結果']; if (res !== '○' && res !== '✕') return;
    const k = d['項目ID'] + '|' + d['項目'];
    const o = it[k] || (it[k] = { cat: d['カテゴリ'], sub: d['サブカテゴリ'], text: d['項目'], ng: 0, n: 0, stores: {} });
    o.n++;
    if (res === '✕') { o.ng++; o.stores[d['店舗']] = (o.stores[d['店舗']] || 0) + 1; }
  });
  const ngRows = Object.keys(it).map(k => it[k]).filter(o => o.ng > 0)
    .sort((a, b) => (b.ng - a.ng) || (b.ng / b.n - a.ng / a.n)).slice(0, 20)
    .map(o => [o.text, Object.keys(o.stores).sort((x, y) => o.stores[y] - o.stores[x]).slice(0, 3).map(s => s + (o.stores[s] > 1 ? '（' + o.stores[s] + '）' : '')).join('、'),
      shortCat_(o.cat), o.sub, o.ng, o.n, o.ng / o.n]);

  // チェック者別
  const byC = {};
  recs.forEach(r => { const c = r['チェック者'] || '（未入力）'; (byC[c] = byC[c] || []).push(r); });
  const chkRows = Object.keys(byC).map(c => {
    const rs = byC[c], m = avg_(rs.map(r => num_(r['得点率'])));
    return [c, rs.length, m == null ? '' : m, (m != null && allMean != null) ? m - allMean : '', avg_(rs.map(r => num_(r['✕'])))];
  }).sort((a, b) => b[1] - a[1]);

  // 書き出し
  sh.getRange(1, 1).setValue('QSCチェック ダッシュボード').setFontSize(16).setFontWeight('bold').setFontColor('#1a1a2e');
  sh.getRange(2, 1).setValue('最終更新 ' + Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm') + '　／　点検 ' + recs.length + '件　／　店舗 ' + storeRows.length + '店　／　全体平均 ' + (allMean == null ? '--' : (allMean * 100).toFixed(1) + '%'))
    .setFontColor('#666666');
  let r = 4;
  const section = (title, note, header, rows) => {
    sh.getRange(r, 1).setValue(title).setFontSize(12).setFontWeight('bold').setFontColor('#e94560');
    sh.getRange(r + 1, 1).setValue(note).setFontSize(9).setFontColor('#999999');
    const hr = r + 2;
    sh.getRange(hr, 1, 1, header.length).setValues([header]).setBackground('#16213e').setFontColor('#ffffff').setFontWeight('bold').setVerticalAlignment('middle');
    sh.setRowHeight(hr, 26);
    if (rows.length) sh.getRange(hr + 1, 1, rows.length, header.length).setValues(rows).setVerticalAlignment('middle');
    else sh.getRange(hr + 1, 1).setValue('まだデータがありません').setFontColor('#999999');
    const top = hr + 1, n = Math.max(rows.length, 1);
    if (rows.length) sh.getRange(top, 1, n, header.length).setBorder(null, null, true, null, null, true, '#eeeeee', SpreadsheetApp.BorderStyle.SOLID);
    r = top + n + 2;
    return { top, n, has: rows.length > 0 };
  };

  const sHead = ['店舗', '業態', '点検数', '平均得点率', '判定'].concat(catNames.map(shortCat_), ['最新チェック日', '最新の点', '前回比']);
  const s = section('店舗別スコア', '平均得点率の高い順。点検数が多い店ほど数字が安定します。判定：' + JUDGE.map(j => j[1] + (j[0] ? ' ' + Math.round(j[0] * 100) + '%以上' : '')).join('／'), sHead, storeRows);
  if (s.has) {
    const K = catNames.length;
    const pctCols = [4].concat(catNames.map((_, i) => 6 + i), [7 + K]);
    pctCols.forEach(c => { const rg = sh.getRange(s.top, c, s.n, 1); rg.setNumberFormat('0.0%'); rules.push(pctRule_(rg)); });
    const jr = sh.getRange(s.top, 5, s.n, 1);
    rules.push(textRule_(jr, '◎', '#d4edda', '#1e7b45'), textRule_(jr, '○', '#e8f5e9', '#2e7d32'), textRule_(jr, '△', '#fff3cd', '#8a6200'), textRule_(jr, '✕', '#fbd9d6', '#c23b33'));
    const dr = sh.getRange(s.top, 8 + K, s.n, 1);
    dr.setNumberFormat('+0.0%;-0.0%;0.0%');
    rules.push(SpreadsheetApp.newConditionalFormatRule().whenNumberGreaterThan(0).setFontColor('#1e7b45').setRanges([dr]).build(),
      SpreadsheetApp.newConditionalFormatRule().whenNumberLessThan(0).setFontColor('#c23b33').setRanges([dr]).build());
    sh.getRange(s.top, 1, s.n, 1).setFontWeight('bold');
    sh.getRange(s.top, 3, s.n, 3 + K).setHorizontalAlignment('center');
  }

  const n2 = section('✕が多い項目 TOP20', '全店の点検で✕がついた回数の多い順。「評価数」は○か✕をつけた回数。', ['項目', '✕が多い店舗', 'カテゴリ', 'サブカテゴリ', '✕の数', '評価数', '✕率'], ngRows);
  if (n2.has) {
    sh.getRange(n2.top, 1, n2.n, 2).setWrap(true);
    const pr = sh.getRange(n2.top, 7, n2.n, 1); pr.setNumberFormat('0%');
    rules.push(SpreadsheetApp.newConditionalFormatRule().setGradientMinpointWithValue('#ffffff', SpreadsheetApp.InterpolationType.NUMBER, '0').setGradientMaxpointWithValue('#f4a9a3', SpreadsheetApp.InterpolationType.NUMBER, '1').setRanges([pr]).build());
    sh.getRange(n2.top, 3, n2.n, 5).setHorizontalAlignment('center');
    sh.getRange(n2.top, 5, n2.n, 1).setFontWeight('bold').setFontColor('#c23b33');
  }

  const n3 = section('チェック者別', '点が甘い・辛いの目安。「全体との差」がプラスなら甘め、マイナスなら辛め。', ['チェック者', '点検数', '平均得点率', '全体との差', '1回あたりの✕'], chkRows);
  if (n3.has) {
    sh.getRange(n3.top, 3, n3.n, 1).setNumberFormat('0.0%');
    sh.getRange(n3.top, 4, n3.n, 1).setNumberFormat('+0.0%;-0.0%;0.0%');
    sh.getRange(n3.top, 5, n3.n, 1).setNumberFormat('0.0');
    sh.getRange(n3.top, 2, n3.n, 4).setHorizontalAlignment('center');
  }

  sh.setConditionalFormatRules(rules);
  widths_(sh, [300, 190, 110, 90, 90, 80, 80, 80, 80, 80, 105, 80, 80]);
  sh.setFrozenRows(0);
}

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const head = (sh, n) => { sh.getRange(1, 1, 1, n).setBackground('#16213e').setFontColor('#ffffff').setFontWeight('bold'); sh.setFrozenRows(1); };
  const col = (sh, c) => sh.getRange(2, c, Math.max(sh.getMaxRows() - 1, 1), 1);

  let sh = ss.getSheetByName(SHEET.RECORDS);
  head(sh, HEADERS.RECORDS.length);

    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('✕').setBackground('#fbd9d6').setFontColor('#c23b33').setBold(true).setRanges([jr]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('○').setBackground('#e3f4ea').setFontColor('#1e7b45').setRanges([jr]).build(),

  sh = ss.getSheetByName(SHEET.STORES);
  head(sh, HEADERS.STORES.length);

  let dash = ss.getSheetByName(SHEET.DASH);
  if (!dash) dash = ss.insertSheet(SHEET.DASH, 0);
  ss.setActiveSheet(dash); ss.moveActiveSheet(1);

  if (s1 && s1.getLastRow() === 0 && ss.getSheets().length > 1) ss.deleteSheet(s1);
}

/** Webダッシュボード用のデータ。スクリプトプロパティ VIEW_PIN を設定するとPINが必要になる */
function getDashData(pin) {
  const want = PropertiesService.getScriptProperties().getProperty('VIEW_PIN');
  if (want && String(pin || '') !== String(want)) return { needPin: true, wrong: !!pin };
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const recs = readRows_(ss.getSheetByName(SHEET.RECORDS)).filter(r => r['記録ID']).map(r => [
    String(r['記録ID']), fmtDate_(r['チェック日']), String(r['店舗']), String(r['業態'] || ''), String(r['ブランド'] || ''), String(r['チェック者'] || ''),
    num_(r['得点率']), Number(r['○']) || 0, Number(r['✕']) || 0, Number(r['対象外']) || 0, Number(r['未チェック']) || 0,
    String(r['総評'] || ''), String(r['写真フォルダ'] || ''),
    r['送信日時'] instanceof Date ? Utilities.formatDate(r['送信日時'], 'Asia/Tokyo', 'HH:mm:ss') : '', Number(r['△']) || 0]);
  const cats = readRows_(ss.getSheetByName(SHEET.CATS)).filter(r => r['記録ID']).map(r => [String(r['記録ID']), String(r['カテゴリ']), num_(r['得点率'])]);
  const items = [], idx = {}, res = {};
  const sh = ss.getSheetByName(SHEET.DETAIL);
  if (sh.getLastRow() > 1) {
    sh.getRange(2, 1, sh.getLastRow() - 1, 11).getValues().forEach(v => {
      const rid = String(v[0]); if (!rid) return;
      const key = String(v[6]) + '|' + String(v[7]);
      let i = idx[key];
      if (i == null) { i = idx[key] = items.length; items.push([String(v[6]), String(v[4]), String(v[5]), String(v[7])]); }
      const o = res[rid] || (res[rid] = { p: [], f: [], h: [] });
      if (v[9] === '○') o.p.push(i);
      else if (v[9] === '△') o.h.push([i, String(v[10] || '')]);
      else if (v[9] === '✕') o.f.push([i, String(v[10] || '')]);
    });
  }
  return { now: Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm'), judge: JUDGE, recs, cats, items, res };
}

/* ───────── helpers ───────── */

/** シートの見出しを読み、足りない列があれば右端に足して返す */
function headersFor_(sh, want) {
  const n = Math.max(sh.getLastColumn(), 1);
  const h = sh.getRange(1, 1, 1, n).getValues()[0].map(String);
  while (h.length && h[h.length - 1] === '') h.pop();
  want.forEach(k => { if (h.indexOf(k) < 0) { h.push(k); sh.getRange(1, h.length).setValue(k).setFontWeight('bold'); } });
  return h;
}

function readRows_(sh) {
  if (!sh || sh.getLastRow() < 2) return [];
  const v = sh.getRange(1, 1, sh.getLastRow(), sh.getLastColumn()).getValues();
  const h = v.shift();
  return v.map(row => Object.fromEntries(h.map((k, i) => [k, row[i]])));
}

function fmtDate_(d) {
  if (d instanceof Date) return Utilities.formatDate(d, 'Asia/Tokyo', 'yyyy-MM-dd');
  return String(d || '');
}

/* ───────── 外部ページ（GitHub Pages）からの呼び出し口 ───────── */
// 画面をGitHubに置いた時は、ここ経由でスプシとやり取りする（中身は google.script.run と同じ関数）
const API_FUNCS = { getConfig, saveMaster, updateItem, checkPin, uploadPhoto, submitRecord, getHistory, getDashData };

function doPost(e) {
  let out;
  try {
    const req = JSON.parse(e.postData.contents);
    const f = API_FUNCS[req.fn];
    if (!f) throw new Error('不明な呼び出しです: ' + req.fn);
    out = { ok: true, result: f.apply(null, req.args || []) };
  } catch (err) {
    out = { ok: false, error: String((err && err.message) || err) };
  }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}
